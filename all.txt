let raycaster = new THREE.Raycaster();
let mouse = new THREE.Vector2();
renderer.domElement.addEventListener("click", function(event) {
    mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
    mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
    raycaster.setFromCamera(mouse, camera);
    let hits = raycaster.intersectObjects(cubes);
    if(hits.length > 0) {
        let cube = hits[0].object;
        let currentColor = cube.material.color.getHex();
        let index = colors.findIndex(function(color) {
            return new THREE.Color(color).getHex() === currentColor;
        });
        index = (index + 1) % colors.length;
        cube.material.color.set(colors[index]);
        let win = true;
        cubes.forEach(function(c) {
            if(c.material.color.getHex() !== target.material.color.getHex()) {
                win = false;
            }  });
        if(win) {
            document.getElementById("message").innerHTML = "You Win!";
            clearInterval(timer);
        } }
});

let score = 0;
let raycaster = new THREE.Raycaster();
renderer.domElement.addEventListener("click", function(event) {
    let mouse = new THREE.Vector2();
    mouse.x = (event.clientX / window.innerWidth) * 2 - 1;
    mouse.y = -(event.clientY / window.innerHeight) * 2 + 1;
    raycaster.setFromCamera(mouse, camera);
    let hits = raycaster.intersectObjects(targets);
    if(hits.length > 0) {
        let target = hits[0].object;
        scene.remove(target);
        targets.splice(targets.indexOf(target), 1);
        score++;
        document.getElementById("score").innerHTML ="Score: " + score;
    }
});

let health = 100;
let touching = false;
const value = document.getElementById("value");
const bar = document.getElementById("bar");
const message = document.getElementById("message");
let keys = {};
window.addEventListener("keydown", function(event) {
    keys[event.key] = true;
});
window.addEventListener("keyup", function(event) {
    keys[event.key] = false;

function animate() {
    requestAnimationFrame(animate);
    if (health > 0) {
        if (keys["ArrowLeft"]) player.position.x -= 0.05;
        if (keys["ArrowRight"]) player.position.x += 0.05;
        if (keys["ArrowUp"]) player.position.z -= 0.05;
        if (keys["ArrowDown"]) player.position.z += 0.05;
        let distance = player.position.distanceTo(enemy.position);
        if (distance < 1.2 && touching === false) {
            health = health - 10;
            touching = true;
            value.innerHTML = health;
            bar.style.width = (health * 2) + "px";
        }
        if (distance >= 1.2) {
            touching = false;
        }
        if (health <= 0) {
            health = 0;
            value.innerHTML = health;
            bar.style.width = "0px";
            message.innerHTML = "Game Over";
        }
    }
    renderer.render(scene, camera);
}

);

scene.add(player);
const coin = new THREE.Mesh(
    new THREE.SphereGeometry(0.3, 16, 16),
    new THREE.MeshBasicMaterial({ color: "yellow" })
);
coin.position.set(2, 0.4, 0);
scene.add(coin);
let score = 0;
let collected = false;
const scoreText = document.getElementById("score");
const message = document.getElementById("message");
let keys = {};
window.addEventListener("keydown", function(event) {
    keys[event.key] = true;
    if (event.code === "Space" && player.position.y <= 0.4) {
        player.userData.velocity = 0.15;
    }
});
window.addEventListener("keyup", function(event) {
    keys[event.key] = false;
});
let gravity = 0.008;
function animate() {
    requestAnimationFrame(animate);
    if (keys["ArrowLeft"]) player.position.x -= 0.05;
    if (keys["ArrowRight"]) player.position.x += 0.05;
    if (keys["ArrowUp"]) player.position.z -= 0.05;
    if (keys["ArrowDown"]) player.position.z += 0.05;
    let velocity = player.userData.velocity || 0;
    player.position.y += velocity;
    velocity -= gravity;
    player.userData.velocity = velocity;
    if (player.position.y <= 0.4) {
        player.position.y = 0.4;
        player.userData.velocity = 0;
    }
    if (!collected && player.position.distanceTo(coin.position) < 0.8) {
        score++;
        scoreText.innerHTML = "Score: " + score;
        scene.remove(coin);
        collected = true;
    }
    if (player.position.y < -2) {
        message.innerHTML = "Try Again!";
        player.position.set(0, 0.4, 0);
        player.userData.velocity = 0;
        scene.add(coin);
        coin.position.set(2, 0.4, 0);
        collected = false;
        score = 0;
        scoreText.innerHTML = "Score: 0";
        setTimeout(function() {
            message.innerHTML = "";
        }, 1500);
    }
    renderer.render(scene, camera);
}
animate();

//rectangular blocks chain reaction
const ground = new THREE.Mesh(
    new THREE.BoxGeometry(12, 0.2, 5),
    new THREE.MeshBasicMaterial({ color: "green" })
);
ground.position.y = -0.1;
scene.add(ground);
let blocks = [];
for (let i = 0; i < 5; i++) {
    const block = new THREE.Mesh(
        new THREE.BoxGeometry(0.5, 2, 0.5),
        new THREE.MeshBasicMaterial({ color: "orange" })
    );
    block.position.set(-3 + i * 1.5, 1, 0);
    scene.add(block);
    blocks.push(block);
}
const ball = new THREE.Mesh(
    new THREE.SphereGeometry(0.5, 16, 16),
    new THREE.MeshBasicMaterial({ color: "blue" })
);
ball.position.set(-5, 0.5, 0);
scene.add(ball);
const message = document.getElementById("message");
let started = false;
let speed = 0.04;
let keys = [];
window.addEventListener("keydown", function(event) {
    keys[event.key] = true;
});
window.addEventListener("keyup", function(event) {
    keys[event.key] = false;
});
function animate() {
    requestAnimationFrame(animate);
    if (keys["ArrowRight"]) {
        ball.position.x += speed;
    }
    if (keys["ArrowLeft"]) {
        ball.position.x -= speed;
    }
    if (
        ball.position.distanceTo(blocks[0].position) < 1.5 &&
        started === false
    ) {
        started = true;
        message.innerHTML = "Chain Reaction Started!";
    }
    if (started) {
        for (let i = 0; i < blocks.length; i++) {
            blocks[i].rotation.z += 0.04;
            blocks[i].position.y -= 0.02;
            if (blocks[i].position.y < 0.5) {
                blocks[i].position.y = 0.5;
            }
        }}
    renderer.render(scene, camera);
}
animate();
